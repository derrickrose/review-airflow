RTS24 Regulatory Reporting - Market Events Data Processing Pipeline

Orchestrates daily regulatory report generation across five market segments (EQU, ETF, FXI, WAR, BLK),
processing trading data for European regulatory compliance (RTS24).

Purpose:
    Generates RTS24-compliant reports by processing market event data (orders, quotes, rejections,
    market status, quote requests) for regulatory submission to ESMA.

Trigger Conditions:
    - Asset-based: Waits for completion of all segment data processing and Saturn SLC availability
    - Manual: Accepts trading_date parameter for historical data regeneration

Key Features:
    - Parallel processing across market segments for optimal performance
    - Per-segment validation ensuring all event types complete before file generation
    - On-demand file regeneration based on database requests for data corrections
    - Cross-asset date validation preventing data inconsistency

Architecture:
    1. Extract and validate trading_date from triggers
    2. Load market event data per segment (6 flows each: order, reject_order, quote, etc.)
    3. Generate segment-specific regulatory files after data loads complete
    4. Process on-demand regeneration requests with optional filters (MIC, ISIN, participant)
"""

def log_section(title: str) -> None:

def raise_logged_value_error(prefix: str, message: str) -> None:
    """Log a formatted error section and raise a ValueError."""
    logger.info(LOG_SEPARATOR)
    logger.info(f"{prefix}: {message}")
    logger.info(LOG_SEPARATOR)
    raise ValueError(message)


def is_valid_date(date_str: str) -> bool:
    """Check if date string matches expected format (YYYY-MM-DD)."""
    try:
        datetime.strptime(date_str, DATE_FORMAT)
        return True
    except (ValueError, TypeError):
        return False


def extract_trading_date_from_manual_trigger_conf(conf: Dict[str, Any]) -> str:
    """Extract and validate trading_date from manual dag_run.conf."""
    if TRADING_DATE not in conf:
        raise ValueError(
            f"Manual trigger requires '{TRADING_DATE}' in dag_run.conf. "
            f'Example: {{"{{TRADING_DATE}}": "2026-03-20"}}'
        )

    trading_date = conf.get(TRADING_DATE)
    if not trading_date:
        raise ValueError(
            f"Manual trigger '{TRADING_DATE}' cannot be empty. Expected format: {DATE_FORMAT}"
        )

    if not is_valid_date(trading_date):
        raise ValueError(
            f"Invalid '{TRADING_DATE}' format: {trading_date}. Expected {DATE_FORMAT}"
        )

    return trading_date


def parse_event_body(body: Any) -> Optional[Dict[str, Any]]:
    """Parse event body which may be a JSON string or already a dict.

    Handles both formats to support different event sources and API versions.
    Returns None on failure to allow graceful degradation in event processing.
    """
    if isinstance(body, dict):
        return body
    if not isinstance(body, str):
        logger.debug(f"Unexpected Body type in asset event: {type(body)}")
        return None

    try:
        return json.loads(body)
    except (
        ValueError,
        json.JSONDecodeError,
    ) as exc:  # JSONDecodeError subclasses ValueError
        logger.warning(f"Failed to parse asset event Body JSON: {exc} -- Body: {body}")
        return None


def extract_date_from_event(event: Any) -> Optional[str]:
    """Extract `trading_date` from a single asset event (robust to JSON/body errors)."""
    extra = getattr(event, "extra", {}) or {}

    # Try direct path first for performance
    trading_date = extra.get(TRADING_DATE)
    if trading_date:
        return trading_date

    # Fallback to nested structure for legacy event format
    payload = extra.get("payload", {}) or {}
    message_batch = payload.get("message_batch", [])
    if not message_batch:
        return None

    body = message_batch[0].get("Body")
    if not body:
        return None

    parsed = parse_event_body(body)
    if not parsed:
        return None

    trading_date = parsed.get(TRADING_DATE)
    return trading_date


def extract_trading_date_from_asset_events(
    triggering_asset_events: Dict[Any, List[Any]],
) -> str:
    """Extract and validate trading_date across all triggering asset events.

    When multiple assets trigger this DAG, ensures all assets carry the same
    trading_date. Raises if dates conflict or none found.
    """
    dates_by_asset: Dict[str, str] = {}

    for asset_key, asset_events in triggering_asset_events.items():
        for event in asset_events or []:
            trading_date = extract_date_from_event(event)
            if trading_date:
                asset_name = getattr(asset_key, "name", str(asset_key))
                dates_by_asset[asset_name] = trading_date
                break  # Performance optimization: single date per asset sufficient for validation

    if not dates_by_asset:
        raise ValueError("No trading_date found in any triggering asset event")

    unique_dates = set(dates_by_asset.values())
    if len(unique_dates) > 1:
        raise ValueError(
            f"Conflicting trading_date across triggering assets: {dates_by_asset}. "
            f"All assets must carry the same trading_date."
        )

    trading_date = unique_dates.pop()
    logger.info(
        f"Trading date {trading_date} confirmed across {len(dates_by_asset)} asset(s): {list(dates_by_asset.keys())}"
    )
    return trading_date


@task
def extract_trading_date(**context) -> str:
    """Resolve trading date from manual conf first, otherwise from triggering asset events."""
    dag_run = context.get("dag_run")
    conf = dag_run.conf if dag_run else {}
    run_type = dag_run.run_type.value if dag_run else "unknown"

    logger.info(LOG_SEPARATOR)
    logger.info(
        f"Determining trading date for RTS24 report generation (trigger: {run_type})"
    )
    logger.info(LOG_SEPARATOR)

    if run_type == "manual":
        logger.info("Manual trigger detected - validating user-provided trading_date")
        logger.info(f"Configuration input: {conf}")

        try:
            trading_date = extract_trading_date_from_manual_trigger_conf(conf)
            logger.info(
                f"Manual trigger validated - will process data for {trading_date}"
            )
        except ValueError as exc:
            raise_logged_value_error("VALIDATION_ERROR", str(exc))

    else:
        logger.info(
            "Scheduled trigger detected - extracting trading_date from upstream asset events"
        )
        triggering_asset_events = context.get("triggering_asset_events", {})

        try:
            trading_date = extract_trading_date_from_asset_events(
                triggering_asset_events
            )
            if not is_valid_date(trading_date):
                raise ValueError(
                    f"Invalid '{TRADING_DATE}' from triggering asset events: {trading_date}. "
                    f"Expected {DATE_FORMAT}"
                )
            logger.info(
                f"Scheduled trigger validated - will process data for {trading_date}"
            )
        except ValueError as exc:
            raise_logged_value_error("EXTRACTION_ERROR", str(exc))

    logger.info(LOG_SEPARATOR)
    logger.info(f"Trading date confirmed: {trading_date}")
    logger.info(
        f"Scope: Processing 5 market segments (EQU, ETF, FXI, WAR, BLK) with 6 event flows each"
    )
    logger.info(LOG_SEPARATOR)
    return trading_date


def log_emr_job_start(context: Dict[str, Any]) -> None:
    """Log EMR submission details at operator runtime."""
    task = context["task"]
    spark_submit = task.job_driver.get("sparkSubmit", {})
    args = spark_submit.get("entryPointArguments", [])
    entry_point = spark_submit.get("entryPoint", "")

    # Extract business context from task_id and arguments
    task_id = task.task_id
    flow_name = next(
        (args[i + 1] for i, arg in enumerate(args) if arg == "--flow-name"),
        "generation",
    )
    trading_date = next(
        (args[i + 1] for i, arg in enumerate(args) if arg == "--trading-date"),
        "unknown",
    )

    # Determine job type for business context
    if "trd_order" in flow_name:
        job_type = "order data processing"
    elif "trd_quote" in flow_name:
        job_type = "quote data processing"
    elif "trd_market_status" in flow_name:
        job_type = "market status processing"
    elif "trd_reject" in flow_name:
        job_type = "rejection data processing"
    elif "generation" in task_id or "generation" in entry_point:
        job_type = "RTS24 file generation"
    else:
        job_type = "data processing"

    logger.info(LOG_SEPARATOR)
    logger.info(f"Submitting EMR Serverless job: {task_id}")
    logger.info(f"Purpose: {job_type} for RTS24 regulatory reporting")
    logger.info(f"Trading date: {trading_date} | Expected duration: 5-15 minutes")
    logger.info(f"Spark config: {spark_submit.get('sparkSubmitParameters', 'default')}")
    logger.info(f"Job arguments: {args}")
    logger.info(LOG_SEPARATOR)


def build_emr_job_params(
    flow_name: str,
    segment_code: str,
    segment_id: str,
    trading_date: str,
    job_name: str,
) -> Dict[str, Any]:
    """Build EMR job parameters for one flow and one segment.

    Separates job_name from flow_name to support job refactoring without
    breaking existing flow references in the codebase.
    """
    job_base_path = f"{JOBS_BASE_PATH}/{job_name}"

    # Consistent naming: rts24-load-{job_name}-{segment_code}-{trading_date}
    emr_job_name = f"rts24-load-{job_name}-{segment_code}-{trading_date}"

    return {
        "name": emr_job_name,
        "job_driver": {
            "sparkSubmit": {
                "entryPoint": f"{job_base_path}/{JOB_ENTRYPOINT_FILE}",
                "sparkSubmitParameters": SPARK_SUBMIT_PARAMETERS,
                "entryPointArguments": [
                    "--trading-date",
                    trading_date,
                    "--env",
                    ENV_LOWER,
                    "--config",
                    f"{job_base_path}/{JOB_CONFIG_FILE}",
                    "--flow-name",
                    flow_name,
                    "--optiq-segment-id",
                    segment_id,
                    "--log-level",
                    LOG_LEVEL,
                ],
            }
        },
    }


def build_file_generation_emr_job(
    job_name: str,
    segment_code: str,
    segment_id: str,
    trading_date_arg: Optional[str] = None,
) -> Dict[str, Any]:
    """Build EMR job params for the file-generation job.

    Accepts Jinja templates for trading_date to avoid parse-time XCom dependencies
    that would prevent proper task graph construction.
    """
    job_base_path = f"{JOBS_BASE_PATH}/{job_name}"

    args: List[str] = [
        "--env",
        ENV_LOWER,
    ]

    if trading_date_arg:
        args.extend(["--trading-date", trading_date_arg])

    args.extend(
        [
            "--config",
            f"{job_base_path}/{JOB_CONFIG_FILE}",
            "--sql-base-path",
            f"{job_base_path}/{SQL_BASE_SUBPATH}",
            "--optiq-segment-id",
            segment_id,
        ]
    )

    # Consistent naming: rts24-generate-file-{job_name}-{segment_code}-{trading_date}
    if trading_date_arg and "{{" not in str(trading_date_arg):
        emr_job_name = (
            f"rts24-generate-file-{job_name}-{segment_code}-{trading_date_arg}"
        )
    else:
        emr_job_name = f"rts24-generate-file-{job_name}-{segment_code}"

    return {
        "name": emr_job_name,
        "job_driver": {
            "sparkSubmit": {
                "entryPoint": f"{job_base_path}/{JOB_ENTRYPOINT_FILE}",
                "sparkSubmitParameters": SPARK_SUBMIT_PARAMETERS,
                "entryPointArguments": args,
            },
        },
    }


def create_task_group(segment_code: str, segment_id: str) -> tuple:
    """Create a task group for segment processing.

    Returns individual task references to work around Airflow's unreliable
    TaskGroup dependency resolution when using bitshift operators.

    Uses Jinja template for trading_date to resolve at runtime via XCom.
    """
    application_id_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_APPLICATION_ID') }}"
    )
    execution_role_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_ROLE') }}"
    )

    # Use Jinja template to pull trading_date from XCom at runtime
    trading_date_template = "{{ ti.xcom_pull(task_ids='extract_trading_date') }}"

    tasks = []
    with TaskGroup(group_id=f"load_segment_{segment_code}") as task_group:
        for flow_name, job_name in EVENT_FLOWS_BY_SEGMENT.get(segment_code, []):
            task = EmrServerlessStartJobOperator(
                task_id=f"run_{flow_name}_emr_job",
                map_index_template="{{ task.name }}",
                **build_emr_job_params(
                    flow_name, segment_code, segment_id, trading_date_template, job_name
                ),
                application_id=application_id_template,
                execution_role_arn=execution_role_template,
                deferrable=True,
                on_execute_callback=log_emr_job_start,
                config={"tags": EMR_CONFIG_TAGS},
                **EMR_RETRY_POLICY,
            )
            tasks.append(task)

    return task_group, tasks


def create_file_generation_task(segment_code: str, segment_id: str):
    """Create the file generation task outside the TaskGroup.

    Uses runtime XCom resolution to avoid parse-time circular dependencies
    that would prevent proper DAG construction.

    Uses trigger_rule='none_failed' to allow running even if upstream load tasks are skipped.
    """
    application_id_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_APPLICATION_ID') }}"
    )
    execution_role_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_ROLE') }}"
    )

    trading_date_template = "{{ ti.xcom_pull(task_ids='extract_trading_date') }}"

    return EmrServerlessStartJobOperator(
        task_id=f"generate_{segment_code}_files",
        map_index_template="{{ task.name }}",
        **build_file_generation_emr_job(
            "c_reg_8_rts24_generation", segment_code, segment_id, trading_date_template
        ),
        application_id=application_id_template,
        execution_role_arn=execution_role_template,
        trigger_rule='none_failed',  # Run even if upstream load tasks are skipped
        deferrable=True,
        on_execute_callback=log_emr_job_start,
        config={"tags": EMR_CONFIG_TAGS},
        **EMR_RETRY_POLICY,
    )


def create_all_flows():
    """Create one TaskGroup per segment and the file generation task outside.

    Returns individual task references rather than TaskGroups to ensure reliable
    dependency resolution and prevent premature task execution.

    NOTE: Does NOT set dependencies here - dependencies are set in DAG definition
    to allow conditional execution based on ShortCircuit gates.

    NOTE: trading_date is resolved at runtime via Jinja templates in the operators.
    """
    all_load_tasks = []
    generation_tasks = []
    for segment_code, segment_id in SEGMENT_CODE_TO_ID.items():
        task_group, tasks = create_task_group(segment_code, segment_id)
        generation_task = create_file_generation_task(segment_code, segment_id)
        all_load_tasks.extend(tasks)
        generation_tasks.append(generation_task)
    return all_load_tasks, generation_tasks

@task
def process_on_demand_query_results(**context):
    """Process Athena query results and build EMR params for dynamic mapping."""
    ti = context["ti"]
    query_execution_id = ti.xcom_pull(task_ids="query_on_demand_table")

    # For on-demand requests, trading_date comes from each row's data
    trading_date = None  # Will be overridden by row-specific trading_date

    logger.info(LOG_SEPARATOR)
    logger.info(f"Processing on-demand requests query results")
    logger.info(f"Query execution ID: {query_execution_id}")
    logger.info(LOG_SEPARATOR)

    athena_hook = AthenaHook(aws_conn_id=AWS_CONN_ID)

    job_params = []
    request_ids = []  # Collect IDs for later update
    paginator = athena_hook.get_query_results_paginator(query_execution_id)
    is_first_row = True
    request_count = 0

    if paginator:
        for page in paginator:
            result_set = page.get("ResultSet", {})
            result_metadata = result_set.get("ResultSetMetadata", {})
            column_info = result_metadata.get("ColumnInfo", [])
            column_names = [c.get("Name") for c in column_info] if column_info else []
            rows = result_set.get("Rows", [])

            for row in rows:
                # Athena returns column names as first row
                if is_first_row:
                    is_first_row = False
                    continue

                data_cells = row.get("Data", [])
                row_data = {}
                for i, cell in enumerate(data_cells):
                    name = column_names[i] if i < len(column_names) else f"col_{i}"
                    row_data[name] = cell.get("VarCharValue")

                request_count += 1

                # Log business-friendly request interpretation with only present values
                log_parts = [f"Request #{request_count}: Generate RTS24 files"]
                if row_data.get("trading_date"):
                    log_parts.append(f"date={row_data.get('trading_date')}")
                if row_data.get("mic_code"):
                    log_parts.append(f"MIC={row_data.get('mic_code')}")
                if row_data.get("isin_code"):
                    log_parts.append(f"ISIN={row_data.get('isin_code')}")
                if row_data.get("participant_id"):
                    log_parts.append(f"Participant={row_data.get('participant_id')}")

                logger.info(" | ".join(log_parts))
                logger.info(
                    f"Request #{request_count} raw data: {json.dumps(row_data, ensure_ascii=False)}"
                )

                emr_params = build_emr_params_for_row(row_data, trading_date)
                if emr_params:
                    job_params.append(emr_params)
                    logger.info(
                        f"Request #{request_count}: EMR job prepared - {emr_params.get('name', 'unknown')}"
                    )
                    # Collect ID only if EMR job was successfully created
                    request_id = row_data.get("id")
                    if request_id:
                        request_ids.append(request_id)

    logger.info(LOG_SEPARATOR)
    if job_params:
        logger.info(f"Summary: Found {len(job_params)} pending regeneration request(s)")
        logger.info(
            f"Next step: {len(job_params)} EMR Spark jobs will execute in parallel"
        )
        logger.info(f"Expected duration: ~10-15 minutes per job")
    else:
        logger.info("Summary: No pending regeneration requests found (run_flag='0')")
        logger.info("Result: Skipping on-demand file generation - no work to perform")
    logger.info(LOG_SEPARATOR)

    # Store IDs separately for update task
    ti.xcom_push(key="request_ids", value=request_ids)

    return job_params



def build_emr_params_for_row(
    row: Dict[str, Any], trading_date: str
) -> Optional[Dict[str, Any]]:
    """Build EMR operator params for a single Athena row.

    Transforms database records into EMR job parameters for dynamic task mapping,
    enabling on-demand file generation based on user requests.
    """
    if not row:
        return None

    def get_value(column_name: str) -> Optional[str]:
        """Extract and normalize column value from row data."""
        value = row.get(column_name)
        if value is None:
            return None
        if isinstance(value, str):
            stripped = value.strip()
            return stripped if stripped != "" else None
        return str(value)

    # Reuse standard generation job for consistency with scheduled runs
    job_name = "c_reg_8_rts24_generation"
    job_base_path = f"{JOBS_BASE_PATH}/{job_name}"

    # Row-specific date overrides DAG-level for flexibility in regenerating historical data
    row_trading_date = get_value("trading_date")

    # Optional filters for targeted file regeneration
    mic = get_value("mic_code")
    isin = get_value("isin_code")
    participant = get_value("participant_id")

    args: List[str] = ["--env", ENV_LOWER]
    if row_trading_date:
        args.extend(["--trading-date", str(row_trading_date)])
    args.extend(
        [
            "--config",
            f"{job_base_path}/{JOB_CONFIG_FILE}",
            "--sql-base-path",
            f"{job_base_path}/{SQL_BASE_SUBPATH}",
        ]
    )

    if mic:
        args.extend(["--mic-code", str(mic)])
    if isin:
        args.extend(["--isin-code", str(isin)])
    if participant:
        args.extend(["--participant-id", str(participant)])

    job_driver = {
        "sparkSubmit": {
            "entryPoint": f"{job_base_path}/{JOB_ENTRYPOINT_FILE}",
            "sparkSubmitParameters": SPARK_SUBMIT_PARAMETERS,
            "entryPointArguments": args,
        }
    }

    # Build consistent job name: rts24-on-demand-{job_name}-key1_value1-key2_value2-trading_date
    job_name_parts = [f"rts24-on-demand-{job_name}"]
    if mic:
        job_name_parts.append(f"mic_{mic}")
    if isin:
        job_name_parts.append(f"isin_{isin}")
    if participant:
        job_name_parts.append(f"participant_{participant}")
    if row_trading_date:
        job_name_parts.append(row_trading_date)
    name = "-".join(job_name_parts)
    # EMR Serverless enforces 256 character limit
    if len(name) > 256:
        truncated = name[:256]
        # Preserve semantic meaning by truncating at delimiter
        last_sep = max(truncated.rfind("-"), truncated.rfind("_"))
        if last_sep > 0:
            name = truncated[:last_sep]
        else:
            name = truncated

    return {"name": name, "job_driver": job_driver}

@task
def update_ondemand_requests_flag(**context):
    """Update run_flag to 2 for all processed on-demand requests using ID column."""
    ti = context["ti"]

    # Get IDs from the process_on_demand_query_results task
    request_ids = ti.xcom_pull(
        task_ids="process_on_demand_query_results",
        key="request_ids"
    )

    if not request_ids:
        logger.info("No request IDs found to update.")
        return

    athena_hook = AthenaHook(aws_conn_id=AWS_CONN_ID)

    # Single batch UPDATE using IN clause
    # ID column is string type, so escape single quotes and quote values
    escaped_ids = [str(id_val).replace("'", "''") for id_val in request_ids]
    ids_str = ",".join(f"'{id_val}'" for id_val in escaped_ids)

    # Get current date and timestamp for audit columns
    run_date = datetime.now().strftime("%Y-%m-%d")
    run_timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    update_query = (
        f"UPDATE {ONDEMAND_REQUESTS_TABLE} "
        f"SET run_flag = '{RUN_FLAG_COMPLETED}', "
        f"run_date = DATE '{run_date}', "
        f"run_timestamp = '{run_timestamp}' "
        f"WHERE id IN ({ids_str})"
    )

    logger.info(f"Batch updating {len(request_ids)} on-demand requests")
    logger.info(f"  run_flag = '{RUN_FLAG_COMPLETED}'")
    logger.info(f"  run_date = {run_date}")
    logger.info(f"  run_timestamp = {run_timestamp}")
    logger.info(f"Update query: {update_query}")

    athena_hook.run_query(
        query=update_query,
        query_context={"Database": LAKEHOUSE_DATABASE},
        result_configuration={},
        workgroup=ATHENA_WORKGROUP,
    )

    logger.info(f"Update query executed for {len(request_ids)} on-demand requests. Verification will follow.")

@task
def verify_update_status(**context):
    """Verify that the run_flag was successfully updated for all processed requests."""
    ti = context["ti"]

    # Get IDs from the process_on_demand_query_results task
    request_ids = ti.xcom_pull(
        task_ids="process_on_demand_query_results",
        key="request_ids"
    )

    if not request_ids:
        logger.info("No request IDs to verify.")
        return

    athena_hook = AthenaHook(aws_conn_id=AWS_CONN_ID)

    # Build verification query - ID column is string type
    escaped_ids = [str(id_val).replace("'", "''") for id_val in request_ids]
    ids_str = ",".join(f"'{id_val}'" for id_val in escaped_ids)

    verification_query = (
        f"SELECT id, run_flag, run_date, run_timestamp, trading_date, mic_code, isin_code, participant_id "
        f"FROM {ONDEMAND_REQUESTS_TABLE} "
        f"WHERE id IN ({ids_str})"
    )

    logger.info(LOG_SEPARATOR)
    logger.info("Verifying update status for processed requests")
    logger.info(f"Verification query: {verification_query}")
    logger.info(LOG_SEPARATOR)

    query_execution_id = athena_hook.run_query(
        query=verification_query,
        query_context={"Database": LAKEHOUSE_DATABASE},
        result_configuration={},
        workgroup=ATHENA_WORKGROUP,
    )

    # Retrieve and analyze results
    paginator = athena_hook.get_query_results_paginator(query_execution_id)
    is_first_row = True
    verified_count = 0
    failed_updates = []

    if paginator:
        for page in paginator:
            result_set = page.get("ResultSet", {})
            result_metadata = result_set.get("ResultSetMetadata", {})
            column_info = result_metadata.get("ColumnInfo", [])
            column_names = [c.get("Name") for c in column_info] if column_info else []
            rows = result_set.get("Rows", [])

            for row in rows:
                if is_first_row:
                    is_first_row = False
                    continue

                data_cells = row.get("Data", [])
                row_data = {}
                for i, cell in enumerate(data_cells):
                    name = column_names[i] if i < len(column_names) else f"col_{i}"
                    row_data[name] = cell.get("VarCharValue")

                request_id = row_data.get("id")
                run_flag = row_data.get("run_flag")
                run_date = row_data.get("run_date")
                run_timestamp = row_data.get("run_timestamp")

                if run_flag == RUN_FLAG_COMPLETED:
                    verified_count += 1
                    logger.info(
                        f"✓ ID {request_id}: run_flag='{RUN_FLAG_COMPLETED}', "
                        f"run_date={run_date}, run_timestamp={run_timestamp}"
                    )
                else:
                    failed_updates.append(request_id)
                    logger.warning(f"✗ ID {request_id}: run_flag is '{run_flag}' (expected '{RUN_FLAG_COMPLETED}')")

    logger.info(LOG_SEPARATOR)
    logger.info(f"Verification Summary:")
    logger.info(f"  Total requests processed: {len(request_ids)}")
    logger.info(f"  Successfully updated: {verified_count}")
    logger.info(f"  Failed updates: {len(failed_updates)}")

    if failed_updates:
        logger.error(f"  Failed IDs: {failed_updates}")
        raise ValueError(
            f"Update verification failed: {len(failed_updates)} records not updated correctly. "
            f"Failed IDs: {failed_updates}"
        )
    else:
        logger.info("  ✓ All records updated successfully!")

    logger.info(LOG_SEPARATOR)

with DAG(
    dag_id=DAG_ID,
    description=(
        "RTS24 Parameterized Pipeline (ShortCircuit Approach) - Supports multiple execution modes: "
        "full, loads_only, generate_only, ondemand_only"
    ),
    start_date=DAG_START_DATE,
    schedule=reduce(
        and_,
        [
            saturn_slc_ready_asset,
            segment_BLK_ready_asset,
            segment_EQU_ready_asset,
            segment_ETF_ready_asset,
            segment_FXI_ready_asset,
            segment_WAR_ready_asset,
        ],
    ),
    catchup=False,
    max_active_runs=DAG_MAX_ACTIVE_RUNS,
    tags=TAGS + ["parameterized", "shortcircuit"],
    doc_md=__doc__,
):
    # Always extract trading_date (needed for loads and generate modes)
    trading_date = extract_trading_date()

    # Create ShortCircuit gates for different execution paths
    check_loads = should_run_loads()
    check_generate = should_run_generate()
    check_ondemand = should_run_ondemand()

    # Create all tasks (they always exist in DAG structure)
    all_load_tasks, generation_tasks = create_all_flows()

    # On-demand workflow tasks
    query_on_demand_table = AthenaOperator(
        task_id="query_on_demand_table",
        query=f"SELECT id, trading_date, mic_code, isin_code, participant_id FROM {ONDEMAND_REQUESTS_TABLE} WHERE run_flag = '{RUN_FLAG_PENDING}'",
        database=LAKEHOUSE_DATABASE,
        workgroup=ATHENA_WORKGROUP,
        trigger_rule='none_failed',  # Run even if upstream is skipped
        deferrable=True,
        aws_conn_id=AWS_CONN_ID,
    )

    process_on_demand_query_results_task = process_on_demand_query_results()

    application_id_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_APPLICATION_ID') }}"
    )
    execution_role_template = (
        "{{ var.value.get('"
        + ENV_UPPER
        + "_LUNA_CASH_REGULATORY_REPORTING_EMR_ROLE') }}"
    )

    generate_on_demand_files = EmrServerlessStartJobOperator.partial(
        task_id="generate_on_demand_files",
        map_index_template="{{ task.name }}",
        application_id=application_id_template,
        execution_role_arn=execution_role_template,
        deferrable=True,
        on_execute_callback=log_emr_job_start,
        config={"tags": EMR_CONFIG_TAGS},
        **EMR_RETRY_POLICY,
    ).expand_kwargs(process_on_demand_query_results_task)

    update_task = update_ondemand_requests_flag()
    verify_task = verify_update_status()

    # Dependencies with ShortCircuit gates
    # Path 1: Loads (gated by check_loads)
    trading_date >> check_loads >> [*all_load_tasks]

    # Path 2: Generation (gated by check_generate, depends on loads)
    # Generation tasks need both: loads to complete AND check_generate to pass
    trading_date >> check_generate
    for gen_task in generation_tasks:
        [*all_load_tasks] >> gen_task
        check_generate >> gen_task

    # Path 3: On-demand (gated by check_ondemand, depends on loads)
    # On-demand workflow needs: loads to complete AND check_ondemand to pass
    # This matches original: [*all_load_tasks] >> query_on_demand_table >> ...
    trading_date >> check_ondemand
    [*all_load_tasks] >> query_on_demand_table
    check_ondemand >> query_on_demand_table
    query_on_demand_table >> process_on_demand_query_results_task
    process_on_demand_query_results_task >> generate_on_demand_files >> update_task >> verify_task
